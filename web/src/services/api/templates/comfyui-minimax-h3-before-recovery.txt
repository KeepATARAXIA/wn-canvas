function extensionFor(name, type, fallback) {
  const existing = String(name || "").match(/\.([a-z0-9]{1,8})$/i);
  if (existing) return `.${existing[1].toLowerCase()}`;
  const byMime = {
    "image/jpeg": ".jpg",
    "image/png": ".png",
    "image/webp": ".webp",
    "video/mp4": ".mp4",
    "video/quicktime": ".mov",
    "video/webm": ".webm",
    "audio/mpeg": ".mp3",
    "audio/mp4": ".m4a",
    "audio/wav": ".wav",
    "audio/x-wav": ".wav",
    "audio/flac": ".flac",
    "audio/ogg": ".ogg",
  };
  return byMime[String(type || "").toLowerCase()] || fallback;
}

async function uploadInput(blob, originalName, kind, index) {
  const suffix = extensionFor(originalName, blob.type, kind === "image" ? ".png" : kind === "video" ? ".mp4" : ".wav");
  const unique = typeof crypto?.randomUUID === "function" ? crypto.randomUUID() : `${Date.now()}_${Math.random().toString(36).slice(2)}`;
  const filename = `canvas_${kind}_${index + 1}_${unique}${suffix}`;
  const form = new FormData();
  form.append("image", blob, filename);
  form.append("type", "input");
  form.append("subfolder", "infinite-canvas");
  form.append("overwrite", "true");
  let uploaded;
  try {
    uploaded = await request({ method: "post", url: `${comfyUrl}/upload/image`, data: form });
  } catch (error) {
    throw new Error(`上传${kind === "image" ? "图片" : kind === "video" ? "视频" : "音频"}失败：${errorDetail(error)}`);
  }
  if (!uploaded?.name) throw new Error("ComfyUI 上传接口没有返回文件名");
  return [uploaded.subfolder, uploaded.name].filter(Boolean).join("/");
}

async function mediaBlob(media, label) {
  if (media?.blob instanceof Blob) return media.blob;
  if (!media?.url) throw new Error(`${label}没有可读取的文件或 URL`);
  let response;
  try {
    response = await fetch(media.url, { signal });
  } catch (error) {
    throw new Error(`读取${label}失败；如果它是远程 URL，请确认该地址允许跨域读取。${error?.message || error}`);
  }
  if (!response.ok) throw new Error(`读取${label}失败（HTTP ${response.status}）`);
  return response.blob();
}

function closeProgress() {
  stopProgress = true;
  clearTimeout(reconnectTimer);
  socket?.close();
  socket = null;
}

async function connectProgress(reconnecting = false) {
  if (stopProgress || signal?.aborted || typeof onProgress !== "function") return;
  if (typeof WebSocket !== "function") {
    reportStage("无法连接实时进度，正在等待后台结果");
    return;
  }
  await new Promise(resolve => {
    const connection = new WebSocket(comfyUrl.replace(/^http/i, "ws") + "/ws?clientId=" + encodeURIComponent(clientId));
    socket = connection;
    let settled = false;
    const settle = () => { if (!settled) { settled = true; clearTimeout(timer); resolve(); } };
    const timer = setTimeout(() => { settle(); connection.close(); }, 5000);
    connection.addEventListener("open", () => {
      settle();
      if (reconnecting && !stopProgress && !signal?.aborted) reportStage("进度已重连，等待后台更新");
    }, { once: true });
    connection.addEventListener("error", () => { settle(); connection.close(); }, { once: true });
    connection.addEventListener("close", () => {
      settle();
      if (stopProgress || signal?.aborted || socket !== connection) return;
      reportStage("进度连接中断，正在重连（后台任务继续）");
      reconnectTimer = setTimeout(() => { void connectProgress(true); }, 1500);
    }, { once: true });
    connection.addEventListener("message", event => {
      if (stopProgress || signal?.aborted || socket !== connection || typeof event.data !== "string") return;
      let message;
      try { message = JSON.parse(event.data); } catch { return; }
      const data = message?.data;
      if (data?.prompt_id !== promptId) return;
      if (["execution_start", "executing", "progress"].includes(message.type)) executionSeen = true;
      if (message.type === "execution_start") reportStage("后台已开始，正在准备模型");
      if (message.type === "executing") {
        reportStage(data.node == null ? "后台执行结束，正在读取结果" : nodeStage(String(data.node)));
      }
      if (message.type === "progress" && Number.isFinite(data.value) && Number.isFinite(data.max) && data.max > 0) {
        const nodeId = String(data.node);
        if (!workflow[nodeId]) return;
        const percent = Math.max(0, Math.min(100, data.value / data.max * 100));
        const stage = workflow[nodeId].class_type === "SamplerCustomAdvanced"
          ? `正在采样 ${data.value}/${data.max} 步`
          : nodeStage(nodeId);
        reportStage(stage, percent);
      }
      if (message.type === "execution_error") reportStage("后台执行失败，正在读取错误详情");
      if (message.type === "execution_interrupted") reportStage("后台任务已停止");
    });
  });
}
signal?.addEventListener("abort", closeProgress, { once: true });

async function cancelOwnJob() {
  try {
    const response = await fetch(http.url(`${comfyUrl}/api/jobs/${encodeURIComponent(promptId)}/cancel`), {
      method: "POST",
      signal: AbortSignal.timeout(10000),
    });
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
  } catch (error) {
    throw new Error(`无法确认后台任务 ${promptId} 已取消，请在 ComfyUI 队列中检查该任务。${errorDetail(error)}`);
  }
}

async function generate() {
  reportStage("正在连接后台进度，等待提交");
  await connectProgress();
  if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
  if (socket?.readyState === 1) reportStage("正在提交任务");
  let queued;
  try {
    submissionStarted = true;
    const response = await fetch(http.url(`${comfyUrl}/prompt`), {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prompt: workflow, client_id: clientId, prompt_id: promptId }),
      signal: AbortSignal.timeout(30000),
    });
    if (!response.ok) throw new Error(`HTTP ${response.status}：${await response.text()}`);
    queued = await response.json();
  } catch (error) {
    throw new Error(`ComfyUI 拒绝了工作流：${errorDetail(error)}`);
  }

  if (!queued?.prompt_id) throw new Error(`ComfyUI 没有返回任务 ID：${JSON.stringify(queued || {})}`);
  promptId = queued.prompt_id;
  if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
  if (socket?.readyState === 1 && !executionSeen) reportStage("已提交，正在排队等待执行");

  function historyError(entry) {
    const messages = entry?.status?.messages || [];
    const failure = messages.find((item) => item?.[0] === "execution_error")?.[1];
    return failure?.exception_message || failure?.exception_type || entry?.status?.status_str || "未知执行错误";
  }

  function findOutputFile(entry) {
    const output = entry?.outputs?.[OUTPUT_NODE_ID];
    if (!output) return null;
    const files = Object.values(output).flatMap((value) => (Array.isArray(value) ? value : []));
    return (
      files.find((file) => file?.filename && /\.(mp4|mov|webm|mkv)(\?.*)?$/i.test(file.filename)) ||
      files.find((file) => file?.filename && String(file?.format || "").startsWith("video/")) ||
      null
    );
  }

  const outputFile = await poll(
    () => request({ method: "get", url: `${comfyUrl}/history/${encodeURIComponent(promptId)}` }),
    (history) => {
      const entry = history?.[promptId];
      if (!entry) return null;
      if (entry.status?.status_str === "error") throw new Error(`ComfyUI 执行失败：${historyError(entry)}`);
      const file = findOutputFile(entry);
      if (file) return file;
      if (entry.status?.completed === true || entry.status?.status_str === "success") {
        throw new Error(`ComfyUI 已完成，但输出节点 ${OUTPUT_NODE_ID} 没有返回视频文件`);
      }
      return null;
    },
    { intervalMs: 2500, timeoutMs: 3600000 },
  );
  completed = true;
  closeProgress();
  reportStage("正在传回视频到画布");

  const query = new URLSearchParams({
    filename: outputFile.filename,
    subfolder: outputFile.subfolder || "",
    type: outputFile.type || "output",
  });
  const outputBlob = await request({
    method: "get",
    url: `${comfyUrl}/view?${query.toString()}`,
    responseType: "blob",
  });
  return { blob: outputBlob };
}

try {
  return await generate();
} catch (error) {
  // Cancellation uses a fresh signal and an ID-scoped API; never clear the queue
  // or send a global interrupt, which could stop another canvas user's job.
  if (submissionStarted && !completed) await cancelOwnJob();
  if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
  throw error;
} finally {
  closeProgress();
  signal?.removeEventListener("abort", closeProgress);
}
